CREATE VIEW [dbo].[vw_CoursePriceFilter] AS
SELECT CourseId, CourseName, CourseDescription, CoursePrice, CourseType,
       CASE 
           WHEN CoursePrice BETWEEN 20.00 AND 50.00 THEN '20.00 - 50.00'
           WHEN CoursePrice BETWEEN 50.01 AND 100.00 THEN '50.01 - 100.00'
           WHEN CoursePrice BETWEEN 100.01 AND 200.00 THEN '100.01 - 200.00'
           ELSE 'Inny przedział' 
       END AS PriceRange
FROM dbo.Courses;
